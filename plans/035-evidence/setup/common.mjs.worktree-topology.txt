export const json = file => JSON.parse(readFileSync(file, 'utf8'));

export function writeJson(file, value, exclusive = false) {
  mkdirSync(path.dirname(file), { recursive: true });
  writeFileSync(file, JSON.stringify(value, null, 2) + '\n', { flag: exclusive ? 'wx' : 'w' });
}
export const save = (file, value, exclusive = false) => writeJson(path.join(directory, file), value, exclusive);
export const relative = file => path.relative(repo, file).replaceAll('\\', '/');
export function filesUnder(directory) {
  return readdirSync(directory, { withFileTypes: true }).flatMap(entry => {
    assert.ok(!entry.isSymbolicLink(), `No dependency traversal in fingerprints: ${entry.name}`);
    const file = path.join(directory, entry.name);
    return entry.isDirectory() ? filesUnder(file) : [file];
  }).sort();
}
// Exact ordering and path/content concatenation used by the unchanged benchmark.
export function harnessTreeHash(root) {
  const digest = createHash('sha256');
  function visit(directory) {
    for (const entry of readdirSync(directory, { withFileTypes: true }).sort((a,b) => a.name.localeCompare(b.name))) {
      const file = path.join(directory, entry.name);
      if (entry.isDirectory()) visit(file);
      else if (entry.isFile()) digest.update(path.relative(root, file).replaceAll('\\','/')).update(readFileSync(file));
    }
  }
  visit(root);
  return digest.digest('hex');
}
export function fingerprint() {
  const parts = ['packages/vite-plugin-react-docgen-typescript/src', 'packages/vite-plugin-react-docgen-typescript/dist', 'benchmarks/fixtures/large-project', 'benchmarks/fixtures/react-typing'];
  const files = [...parts.flatMap(part => filesUnder(path.join(repo, part))), ...['scripts/benchmark-playground.mjs', 'package.json', 'yarn.lock', 'packages/vite-plugin-react-docgen-typescript/package.json', 'packages/vite-plugin-react-docgen-typescript/build.config.ts'].map(part => path.join(repo,part)).filter(existsSync)];
  return Object.fromEntries(files.sort().map(file => [relative(file), hash(readFileSync(file))]));
}
export function scriptFingerprint() {
  return Object.fromEntries(filesUnder(directory).filter(file => file.endsWith('.mjs')).map(file => [relative(file), hash(readFileSync(file))]));
}
export async function identity() {
  const gitHead = (await run('git', ['rev-parse','HEAD'], { cwd:repo, windowsHide:true })).stdout.trim();
  assert.equal(gitHead,base);
  const dependencies = Object.fromEntries(['typescript','react-docgen-typescript','react','@types/react','vite'].map(name => [name,json(require.resolve(name+'/package.json')).version]));
  return {
    harnessIdentity: { buildSha256:harnessTreeHash(path.dirname(dist)), sourceSha256:harnessTreeHash(path.join(repo,'packages/vite-plugin-react-docgen-typescript/src')), benchmarkSha256:hash(readFileSync(harness)), gitHead, pluginVersion:json(path.join(repo,'packages/vite-plugin-react-docgen-typescript/package.json')).version, dependencies },
    nodeVersion:process.version, executable:process.execPath, executableSha256:hash(readFileSync(process.execPath)), platform:process.platform, os:{ release:os.release(),arch:os.arch() },
    typescript6Alias:json(require.resolve('typescript6/package.json')).version,
    lockfileSha256:hash(readFileSync(path.join(repo,'yarn.lock'))), inputs:fingerprint(),
    scenarios:Object.fromEntries(['large-project','react-typing'].map(name => [name,{ sourceSha256:harnessTreeHash(path.join(repo,'benchmarks/fixtures',name,'src')),tsconfig:json(path.join(repo,'benchmarks/fixtures',name,'tsconfig.json')) }]))
  };
}
export function metadata(result) {
  const code = String(typeof result === 'string' ? result : result?.code ?? '');
  const assignments = [...code.matchAll(/__docgenInfo\s*=\s*(\{[^\r\n]*\})/g)];
  assert.ok(assignments.length,'Missing docgen assignment');
  const docs = assignments.map(match => JSON.parse(match[1])).map(doc => ({
    displayName:doc.displayName,description:doc.description,
    props:Object.fromEntries(Object.entries(doc.props).sort(([a],[b])=>a.localeCompare(b)).map(([name,prop]) => [name,{ name:prop.name, description:prop.description, required:prop.required, type:prop.type, defaultValue:prop.defaultValue }]))
  }));
  return docs.sort((a,b)=>JSON.stringify(a).localeCompare(JSON.stringify(b)));
}
export function observeExtractions() {
  const { Parser } = require(path.join(path.dirname(require.resolve('react-docgen-typescript')),'parser.js'));
  const original = Parser.prototype.getComponentInfo;
  let count = 0;
  Parser.prototype.getComponentInfo = function (...args) { count++; return original.apply(this,args); };
  return () => count;
}
export function cacheCount(directory) { return existsSync(directory) ? filesUnder(directory).filter(file=>file.endsWith('.json')).length : 0; }
export async function launch(args, record, timeout=180000) {
  record.startedAt = new Date().toISOString();
  record.args = args;
  await new Promise((resolve,reject) => {
    const processChild = execFile(process.execPath,args,{cwd:repo,windowsHide:true,timeout,maxBuffer:8000000},(error,stdout,stderr)=>{
      Object.assign(record,{finishedAt:new Date().toISOString(),stdout,stderr,exitCode:error?.code??0});
      if(error) reject(error); else resolve();
    });
    record.childPid=processChild.pid;
  });
}

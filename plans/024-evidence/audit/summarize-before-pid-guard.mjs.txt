import assert from "node:assert/strict";
import { readFileSync, writeFileSync } from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

export const directory = path.dirname(fileURLToPath(import.meta.url));
export const repo = path.resolve(directory, "../..");
export const base = "5f448ec8d596854eace55f59faa669193d187310";
export const states = ["off", "populate", "restart"];
export const groups = ["large-project", "react-typing"].flatMap((scenario) => ["default", "projectService"].map((mode) => ({ scenario, mode })));
export const readJson = (file) => JSON.parse(readFileSync(path.join(directory, file), "utf8"));
export const writeJson = (file, data) => writeFileSync(path.join(directory, file), JSON.stringify(data, null, 2) + "\n");
export const key = ({ scenario, mode }) => `${scenario}/${mode}`;
export const median = (values) => {
  const sorted = [...values].sort((a, b) => a - b);
  const middle = Math.floor(sorted.length / 2);
  return sorted.length % 2 ? sorted[middle] : (sorted[middle - 1] + sorted[middle]) / 2;
};
const stats = (values) => ({ median: median(values), mad: median(values.map((value) => Math.abs(value - median(values)))) });
const finite = (value) => assert.ok(Number.isFinite(value) && value >= 0, `Invalid metric: ${value}`);

export function validateReport(report, expected, reference) {
  assert.equal(report.schemaVersion, 2);
  assert.equal(report.benchmarkKind, "direct-plugin");
  assert.equal(report.cache, expected.cache);
  assert.equal(report.scenario.name, expected.scenario);
  assert.equal(report.scenario.scale, 1);
  assert.equal(report.iterations, 1);
  assert.deepEqual(report.modes, [expected.mode]);
  assert.equal(report.results.length, 1);
  assert.equal(report.results[0].mode, expected.mode);
  assert.equal(report.results[0].runs.length, 1);
  assert.equal(report.identity.gitHead, base);
  assert.deepEqual(report.identity.dependencies, { typescript: "6.0.3", "react-docgen-typescript": "2.2.2", react: "18.3.1", "@types/react": "18.2.25", vite: "8.1.5" });
  assert.equal(report.nodeVersion, "v24.10.0");
  assert.equal(report.platform, "win32");
  for (const name of ["sourceSha256", "buildSha256", "benchmarkSha256"]) assert.match(report.identity[name], /^[a-f0-9]{64}$/);
  assert.match(report.scenario.sourceSha256, /^[a-f0-9]{64}$/);
  assert.ok(Number.isInteger(report.processId) && report.processId > 0);
  if (expected.childPid !== undefined) assert.equal(report.processId, expected.childPid);
  assert.ok(Number.isFinite(Date.parse(report.createdAt)));
  const run = report.results[0].runs[0];
  assert.equal(run.mode, expected.mode);
  assert.equal(run.cache, expected.cache);
  assert.equal(run.processFirstMeasuredInstance, true);
  assert.equal(run.fileCount, report.scenario.fileCount);
  assert.ok(Number.isInteger(run.fileCount) && run.fileCount > 0);
  for (const name of ["coldBatchMs", "configResolvedMs", "firstBatchMs", "warmBatchMs", "sessionTotalMs"]) finite(run[name]);
  assert.ok(Math.abs(run.coldBatchMs - run.configResolvedMs - run.firstBatchMs) < 0.000001);
  assert.equal(run.componentHmr.status, "updated");
  finite(run.componentHmr.totalCycleMs);
  for (const name of ["affectedModuleCount", "invalidatedModuleCount"]) assert.ok(Number.isInteger(run.componentHmr[name]) && run.componentHmr[name] >= 0);
  assert.ok(run.componentHmr.affectedModuleCount > 0);
  const cache = run.cacheLifecycle;
  for (const name of ["initialEntryCount", "finalEntryCount"]) assert.ok(Number.isInteger(cache[name]) && cache[name] >= 0);
  assert.equal(cache.initialEntryCount > 0, expected.cache === "restart");
  assert.equal(cache.finalEntryCount > 0, expected.cache !== "off");
  if (expected.cache === "restart") {
    assert.ok(Number.isInteger(cache.seedProcessId) && cache.seedProcessId > 0);
    assert.notEqual(cache.seedProcessId, report.processId);
  } else assert.equal(cache.seedProcessId, null);
  if (expected.scenario === "react-typing") {
    assert.equal(cache.validation.fixtureValidation.compilerDiagnostics, 0);
    assert.match(cache.validation.fixtureValidation.reactDeclaration, /[\\/]@types[\\/]react[\\/]index\.d\.ts$/);
    assert.notEqual(cache.validation.processId, report.processId);
  }
  if (reference) {
    assert.deepEqual(report.identity, reference.identity);
    assert.equal(report.nodeVersion, reference.nodeVersion);
    assert.equal(report.platform, reference.platform);
    assert.deepEqual(report.timingScope, reference.timingScope);
    if (report.scenario.name === reference.scenario.name) {
      assert.deepEqual(report.scenario, reference.scenario);
      assert.deepEqual(report.parserOptions, reference.parserOptions);
      if (expected.mode === reference.results[0].mode) {
        const other = reference.results[0].runs[0];
        for (const name of ["affectedModuleCount", "invalidatedModuleCount"]) assert.equal(run.componentHmr[name], other.componentHmr[name]);
      }
    }
  }
  return run;
}

export function summarize(manifest) {
  assert.equal(manifest.status, "COMPLETE");
  assert.ok(["initial", "final"].includes(manifest.stage));
  assert.equal(manifest.failures.length, 0, "Failed invocations prevent a verdict");
  assert.deepEqual(manifest.before, manifest.after, "Inputs changed during capture");
  const rows = [];
  const references = new Map();
  const seenPaths = new Set();
  const seenPids = new Set();
  const initialNoisy = new Set();
  let globalReference;
  for (const group of groups) for (const cache of states) {
    const samples = manifest.samples.filter((sample) => key(sample) === key(group) && sample.cache === cache);
    const expectedCount = manifest.extendedGroups.includes(key(group)) ? 10 : 5;
    assert.equal(samples.length, expectedCount, `Missing samples for ${key(group)}/${cache}`);
    assert.deepEqual(samples.map((sample) => sample.round).sort((a, b) => a - b), Array.from({ length: expectedCount }, (_, i) => i + 1));
    const runs = samples.map((sample) => {
      assert.equal(sample.status, "PASS");
      assert.ok(!seenPaths.has(sample.report), "Duplicate report path");
      seenPaths.add(sample.report);
      assert.ok(!seenPids.has(sample.childPid), "Duplicate measured child PID");
      seenPids.add(sample.childPid);
      assert.ok(!path.isAbsolute(sample.report) && !sample.report.split(/[\\/]/).includes(".."));
      const report = readJson(sample.report);
      const run = validateReport(report, sample, globalReference);
      globalReference ??= report;
      validateReport(report, sample, references.get(key(group)));
      references.set(key(group), report);
      return run;
    });
    const first = runs[0];
    const firstFive = samples.map((sample, index) => ({ sample, run: runs[index] })).filter(({ sample }) => sample.round <= 5).map(({ run }) => run);
    if ([stats(firstFive.map((run) => run.coldBatchMs)), stats(firstFive.map((run) => run.componentHmr.totalCycleMs)), stats(firstFive.map((run) => run.sessionTotalMs))].some((metric) => metric.mad > metric.median * 0.2)) initialNoisy.add(key(group));
    rows.push({ ...group, cache, samples: samples.map((sample) => sample.report), count: runs.length,
      cold: stats(runs.map((run) => run.coldBatchMs)), warm: stats(runs.map((run) => run.warmBatchMs)),
      hmr: stats(runs.map((run) => run.componentHmr.totalCycleMs)), session: stats(runs.map((run) => run.sessionTotalMs)),
      fileCount: first.fileCount, affectedModuleCount: first.componentHmr.affectedModuleCount, invalidatedModuleCount: first.componentHmr.invalidatedModuleCount,
      initialEntries: [...new Set(runs.map((run) => run.cacheLifecycle.initialEntryCount))], finalEntries: [...new Set(runs.map((run) => run.cacheLifecycle.finalEntryCount))],
    });
  }
  assert.equal(seenPaths.size, manifest.samples.length, "Unexpected samples");
  if (manifest.stage === "final") assert.deepEqual([...manifest.extendedGroups].sort(), [...initialNoisy].sort(), "Extensions must exactly match first-five variance gates");
  const noisyGroups = [...new Set(rows.filter((row) => [row.cold, row.hmr, row.session].some((metric) => metric.mad > metric.median * 0.2)).map(key))];
  const comparisons = groups.map((group) => {
    const [off, populate, restart] = states.map((cache) => rows.find((row) => key(row) === key(group) && row.cache === cache));
    const coldSavingMs = off.cold.median - restart.cold.median;
    const coldSavingPercent = 100 * coldSavingMs / off.cold.median;
    const populationSessionOverheadMs = populate.session.median - off.session.median;
    const restartSessionSavingMs = off.session.median - restart.session.median;
    const hmr = [populate, restart].map((row) => {
      const deltaMs = row.hmr.median - off.hmr.median;
      const deltaPercent = 100 * deltaMs / off.hmr.median;
      return { cache: row.cache, deltaMs, deltaPercent, materialRegression: deltaMs > 10 && deltaPercent > 10 };
    });
    return { ...group, coldSavingMs, coldSavingPercent, benefitThreshold: coldSavingMs >= 100 && coldSavingPercent >= 20,
      hmr, populationSessionOverheadMs, restartSessionSavingMs,
      projectedReuseSessionsToBreakEven: restartSessionSavingMs > 0 ? Math.ceil(Math.max(0, populationSessionOverheadMs) / restartSessionSavingMs) : null };
  });
  const verdict = noisyGroups.length ? "INCONCLUSIVE" : comparisons.some((comparison) => comparison.benefitThreshold) && !comparisons.some((comparison) => comparison.hmr.some((hmr) => hmr.materialRegression)) ? "KEEP" : "SIMPLIFY_OR_DEPRECATE";
  return { schemaVersion: 1, evaluatedIdentity: globalReference.identity, rows, comparisons, initialNoisyGroups: [...initialNoisy], noisyGroups, verdict,
    scope: "Covered restart and existing-file edit workloads only; initially missing external files remain Plan 023. Direct-plugin HMR, not browser latency. Retained memory unmeasured." };
}

if (process.argv[1] && path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)) {
  const summary = summarize(readJson("capture.json"));
  writeJson("summary.json", summary);
  console.log(JSON.stringify({ verdict: summary.verdict, samples: summary.rows.reduce((n, row) => n + row.count, 0), noisyGroups: summary.noisyGroups }));
}

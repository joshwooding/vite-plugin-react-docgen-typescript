import assert from "node:assert/strict";
import { execFile } from "node:child_process";
import { createHash } from "node:crypto";
import { existsSync, lstatSync, mkdirSync, readFileSync, readdirSync, rmSync, unlinkSync } from "node:fs";
import os from "node:os";
import path from "node:path";
import { base, directory, groups, key, readJson, repo, states, summarize, validateReport, writeJson } from "./summarize.mjs";

const harness = path.join(repo, "scripts/benchmark-playground.mjs");
const hash = (value) => createHash("sha256").update(value).digest("hex");
function files(directory) {
  return readdirSync(directory, { withFileTypes: true }).flatMap((entry) => {
    const file = path.join(directory, entry.name);
    assert.ok(!entry.isSymbolicLink(), `Unexpected symbolic link in fingerprint/storage: ${file}`);
    return entry.isDirectory() ? files(file) : [file];
  }).sort();
}
function fingerprint() {
  const source = path.join(repo, "packages/vite-plugin-react-docgen-typescript/src");
  const trackedInputs = ["packages/vite-plugin-react-docgen-typescript/src", "packages/vite-plugin-react-docgen-typescript/dist", "benchmarks/fixtures/large-project", "benchmarks/fixtures/react-typing"].flatMap((part) => files(path.join(repo, part)));
  return { lockfileSha256: hash(readFileSync(path.join(repo, "yarn.lock"))), benchmarkSha256: hash(readFileSync(harness)),
    sourceSha256: hash(files(source).map((file) => path.relative(source, file) + ":" + hash(readFileSync(file))).join("\n")),
    builtIndexSha256: hash(readFileSync(path.join(repo, "packages/vite-plugin-react-docgen-typescript/dist/index.mjs"))),
    sourceBuildFixturesSha256: hash(trackedInputs.map((file) => path.relative(repo, file) + ":" + hash(readFileSync(file))).join("\n")) };
}
function launch(args, record) {
  return new Promise((resolve, reject) => {
    const child = execFile(process.execPath, args, { cwd: repo, windowsHide: true, timeout: 180_000, maxBuffer: 2_000_000 }, (error, stdout, stderr) => {
      Object.assign(record, { finishedAt: new Date().toISOString(), stdout, stderr, exitCode: error?.code ?? 0 });
      if (error) reject(error); else resolve();
    });
    record.childPid = child.pid;
  });
}
function correctnessGate() {
  const restart = readJson("restart-results.json");
  const watcher = readJson("watcher-results.json");
  assert.deepEqual(restart.identity, watcher.identity);
  assert.equal(restart.identity.evaluatedSha, base);
  assert.equal(restart.identity.versions.typescript6, "6.0.2");
  assert.equal(restart.rows.length, 26);
  assert.equal(watcher.rows.length, 4);
  for (const result of [restart, watcher]) {
    assert.equal(result.verdict, "PASS");
    assert.ok(result.rows.every((row) => row.status === "PASS"));
  }
  assert.ok(watcher.rows.every(({ observed }) => observed.watchedExternalAtStartup && observed.externalEvents.some((event) => event.event === "change") && observed.externalHotHooks.length && observed.controlEvents.length));
  return restart.identity;
}

async function capture() {
  assert.ok(!existsSync(path.join(directory, "capture.json")), "Capture exists; preserve evidence rather than overwrite/rerun");
  const correctnessIdentity = correctnessGate();
  mkdirSync(path.join(directory, "timings"), { recursive: true });
  const manifest = { schemaVersion: 1, status: "RUNNING", stage: "initial", startedAt: new Date().toISOString(), executable: process.execPath, cwd: repo,
    correctnessIdentity, before: fingerprint(), after: null, extendedGroups: [], samples: [], failures: [] };
  assert.equal(manifest.before.lockfileSha256, correctnessIdentity.lockfileSha256);
  assert.equal(manifest.before.benchmarkSha256, correctnessIdentity.benchmarkSha256);
  assert.equal(manifest.before.sourceSha256, correctnessIdentity.sourceSha256);
  assert.equal(manifest.before.builtIndexSha256, correctnessIdentity.buildSha256);
  const references = new Map();
  let firstReport;
  const save = () => writeJson("capture.json", manifest);
  save();
  async function rounds(start, end, selectedGroups) {
    for (let round = start; round <= end; round++) for (const group of selectedGroups) {
      const offset = (round - 1) % states.length;
      const order = [...states.slice(offset), ...states.slice(0, offset)];
      for (const cache of order) {
        const reportFile = `timings/${group.scenario}-${group.mode}-${cache}-${String(round).padStart(2, "0")}.json`;
        const args = [harness, "--scenario", group.scenario, "--scale", "1", "--modes", group.mode, "--cache", cache, "--iterations", "1", "--output", path.join(directory, reportFile)];
        const sample = { invocation: manifest.samples.length + 1, ...group, cache, round, report: reportFile, startedAt: new Date().toISOString(), args, status: "RUNNING" };
        assert.deepEqual(fingerprint(), manifest.before, "Inputs changed before invocation");
        manifest.samples.push(sample);
        try {
          await launch(args, sample);
          const report = readJson(reportFile);
          validateReport(report, sample, firstReport);
          validateReport(report, sample, references.get(key(group)));
          firstReport ??= report;
          references.set(key(group), report);
          assert.ok(!manifest.samples.slice(0, -1).some((previous) => previous.childPid === sample.childPid), "Measured PID reused");
          assert.deepEqual(fingerprint(), manifest.before, "Inputs changed during invocation");
          sample.status = "PASS";
          save();
          console.log(`${sample.invocation}: ${key(group)}/${cache} round ${round}: PASS`);
        } catch (error) {
          sample.status = "FAILED";
          manifest.status = "FAILED";
          manifest.failures.push({ invocation: sample.invocation, message: String(error), stack: error.stack });
          save();
          throw error;
        }
      }
    }
  }
  await rounds(1, 5, groups);
  manifest.after = fingerprint();
  manifest.status = "COMPLETE";
  const initial = summarize(manifest);
  manifest.extendedGroups = initial.noisyGroups;
  if (manifest.extendedGroups.length) {
    manifest.status = "RUNNING";
    save();
    console.log("One bounded extension for: " + manifest.extendedGroups.join(", "));
    await rounds(6, 10, groups.filter((group) => manifest.extendedGroups.includes(key(group))));
  }
  manifest.stage = "final";
  manifest.after = fingerprint();
  manifest.status = "COMPLETE";
  manifest.finishedAt = new Date().toISOString();
  save();
  const summary = summarize(manifest);
  writeJson("summary.json", summary);
  console.log(JSON.stringify({ verdict: summary.verdict, samples: manifest.samples.length, noisyGroups: summary.noisyGroups }));
}

async function storage() {
  assert.ok(!existsSync(path.join(directory, "storage.json")), "Storage evidence already exists");
  const manifest = readJson("capture.json");
  summarize(manifest);
  assert.deepEqual(fingerprint(), manifest.before);
  mkdirSync(path.join(directory, "storage"), { recursive: true });
  const evidence = { measuredDuringTimingMatrix: false, rows: [], failures: [], retainedMemory: null };
  for (const group of groups) {
    const reportFile = `storage/${group.scenario}-${group.mode}-restart.json`;
    const args = [harness, "--scenario", group.scenario, "--scale", "1", "--modes", group.mode, "--cache", "restart", "--iterations", "1", "--output", path.join(directory, reportFile), "--keep-temp"];
    const row = { ...group, cache: "restart", report: reportFile, args, startedAt: new Date().toISOString(), bytes: null, fileCount: null };
    evidence.rows.push(row);
    try {
      await launch(args, row);
      const reference = readJson(manifest.samples.find((sample) => key(sample) === key(group)).report);
      validateReport(readJson(reportFile), row, reference);
      const matches = [...row.stdout.matchAll(/^Kept benchmark workspace at (.+)$/gm)];
      assert.equal(matches.length, 1, "Missing exact temporary workspace path");
      const target = path.resolve(matches[0][1].trim());
      assert.equal(path.dirname(target), path.resolve(os.tmpdir()), "Cleanup outside OS temporary root");
      assert.match(path.basename(target), /^vite-plugin-react-docgen-benchmark-[A-Za-z0-9]{6}$/);
      assert.ok(!lstatSync(target).isSymbolicLink());
      const cacheDirectory = path.join(target, "persistent-cache");
      assert.ok(!lstatSync(cacheDirectory).isSymbolicLink());
      const cacheFiles = files(cacheDirectory);
      row.bytes = cacheFiles.reduce((total, file) => total + lstatSync(file).size, 0);
      row.fileCount = cacheFiles.length;
      row.cacheEntryCount = cacheFiles.filter((file) => file.endsWith(".json")).length;
      // Remove only the fixture's junction itself, never walk through installed dependencies.
      const junction = path.join(target, group.scenario, "node_modules");
      if (existsSync(junction)) {
        assert.ok(lstatSync(junction).isSymbolicLink(), "Unexpected real node_modules in fixture");
        unlinkSync(junction);
      }
      rmSync(target, { recursive: true, force: true });
      row.cleanedTemporaryRoot = target;
      assert.deepEqual(fingerprint(), manifest.before);
      row.status = "PASS";
    } catch (error) {
      row.status = "FAILED";
      evidence.failures.push({ scenario: group.scenario, mode: group.mode, message: String(error) });
      writeJson("storage.json", evidence);
      throw error;
    }
    writeJson("storage.json", evidence);
    console.log(`${key(group)} storage: ${row.bytes} bytes / ${row.fileCount} files`);
  }
}

if (process.argv[2] === "--storage") await storage();
else {
  assert.equal(process.argv[2], "--capture", "Use --capture only after reviewer timing readiness, or --storage after capture");
  await capture();
}
